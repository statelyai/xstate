# The same machine in XState v5

This file is reference material only. It is not compiled and not runnable here — only v6 is installed in this repo. Compare it with [`src/main.ts`](./src/main.ts).

```ts
import { assign, createActor, fromPromise, setup, toPromise } from 'xstate';

const submitFeedback = fromPromise(
  async ({ input }: { input: { rating: number; comment: string } }) => {
    await new Promise((resolve) => setTimeout(resolve, 50));
    return { id: `fb-${input.rating}` };
  }
);

export const feedbackMachine = setup({
  types: {} as {
    context: { rating: number; comment: string; id: string | null };
    events:
      | { type: 'rate'; rating: number }
      | { type: 'comment'; comment: string }
      | { type: 'submit' };
  },
  actors: { submitFeedback },
  guards: {
    isComplete: ({ context }) =>
      context.rating > 0 && context.comment.trim().length > 0
  }
}).createMachine({
  context: { rating: 0, comment: '', id: null },
  initial: 'editing',
  states: {
    editing: {
      on: {
        rate: {
          actions: assign({ rating: ({ event }) => event.rating })
        },
        comment: {
          actions: assign({ comment: ({ event }) => event.comment })
        },
        submit: {
          guard: 'isComplete',
          target: 'submitting'
        }
      }
    },
    submitting: {
      invoke: {
        src: 'submitFeedback',
        input: ({ context }) => ({
          rating: context.rating,
          comment: context.comment
        }),
        onDone: {
          target: 'submitted',
          actions: [
            assign({ id: ({ event }) => event.output.id }),
            ({ event }) => console.log(`submitted as ${event.output.id}`)
          ]
        },
        onError: { target: 'editing' }
      }
    },
    submitted: {
      type: 'final',
      output: ({ context }) => ({ id: context.id })
    }
  }
});
```
